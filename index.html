function S(n, d, l, y) { const fragment = document.createDocumentFragment(); const target = document.getElementById("text"); target.innerHTML = ""; // 前の表示を消す for (let i = 0; i <= n - 1; i++) { if (l[i] == y) { let s = 0; // yの子要素（y+1, y+2...）を探して表示するループ for (let k = 1; s == 0; k++) { let Y = y + k; if (d[Y] !== undefined) { // 子要素が見つかった場合 const li = document.createElement("h1"); const cl = document.createElement("button"); li.textContent = d[Y]; cl.textContent = "下位分類に入る"; cl.addEventListener('click', () => { let j = Y * 10; alert(j + "分類"); S(CATEn, CATEd, CATEl, j); }); fragment.appendChild(li); fragment.appendChild(cl); } else { // これ以上子要素がない場合（定義されていないIDに当たった時） s = 1; const li = document.createElement("h1"); const cl = document.createElement("button"); li.textContent = "以上"; cl.textContent = "上位分類に戻る"; cl.addEventListener('click', () => { // 110 -> 11, 11 -> 1, 1 -> 0 に戻す let j = (y >= 100) ? Math.floor(y / 100) * 10 : 0; alert(j + "分類に戻る"); S(CATEn, CATEd, CATEl, j); }); fragment.appendChild(li); fragment.appendChild(cl); } } break; // 対象の y を処理したら外側のループを抜ける } } target.appendChild(fragment); }
